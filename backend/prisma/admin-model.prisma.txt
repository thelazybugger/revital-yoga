enum AdminRole {
  ADMIN
  STAFF
}

model AdminUser {
  id           Int       @id @default(autoincrement())
  name         String
  email        String    @unique
  passwordHash String
  role         AdminRole @default(ADMIN)
  active       Boolean   @default(true)
  createdAt    DateTime  @default(now())
  updatedAt    DateTime  @updatedAt

  @@index([active])
}
